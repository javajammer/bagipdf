46    pub fn extract_text(&self, page_numbers: &[u32]) -> Result<String> {
47        fn collect_text(text: &mut String, encoding: Option<&str>, operands: &[Object]) {
48            for operand in operands.iter() {
49                match *operand {
50                    Object::String(ref bytes, _) => {
51                        let decoded_text = Document::decode_text(encoding, bytes);
52                        text.push_str(&decoded_text);
53                    }
54                    Object::Array(ref arr) => {
55                        collect_text(text, encoding, arr);
56                        text.push(' ');
57                    }
58                    Object::Integer(i) => {
59                        if i < -100 {
60                            text.push(' ');
61                        }
62                    }
63                    _ => {}
64                }
65            }
66        }
67        let mut text = String::new();
68        let pages = self.get_pages();
69        for page_number in page_numbers {
70            let page_id = *pages.get(page_number).ok_or(Error::PageNumberNotFound(*page_number))?;
71            let fonts = self.get_page_fonts(page_id);
72            let encodings = fonts
73                .into_iter()
74                .map(|(name, font)| (name, font.get_font_encoding()))
75                .collect::<BTreeMap<Vec<u8>, &str>>();
76            let content_data = self.get_page_content(page_id)?;
77            let content = Content::decode(&content_data)?;
78            let mut current_encoding = None;
79            for operation in &content.operations {
80                match operation.operator.as_ref() {
81                    "Tf" => {
82                        let current_font = operation
83                            .operands
84                            .first()
85                            .ok_or_else(|| Error::Syntax("missing font operand".to_string()))?
86                            .as_name()?;
87                        current_encoding = encodings.get(current_font).cloned();
88                    }
89                    "Tj" | "TJ" => {
90                        collect_text(&mut text, current_encoding, &operation.operands);
91                    }
92                    "ET" => {
93                        if !text.ends_with('\n') {
94                            text.push('\n')
95                        }
96                    }
97                    _ => {}
98                }
99            }
100        }
101        Ok(text)
102    }

104    pub fn replace_text(&mut self, page_number: u32, text: &str, other_text: &str) -> Result<()> {
105        let page = page_number.saturating_sub(1) as usize;
106        let page_id = self
107            .page_iter()
108            .nth(page)
109            .ok_or(Error::PageNumberNotFound(page_number))?;
110        let encodings = self
111            .get_page_fonts(page_id)
112            .into_iter()
113            .map(|(name, font)| (name, font.get_font_encoding().to_owned()))
114            .collect::<BTreeMap<Vec<u8>, String>>();
115        let content_data = self.get_page_content(page_id)?;
116        let mut content = Content::decode(&content_data)?;
117        let mut current_encoding = None;
118        for operation in &mut content.operations {
119            match operation.operator.as_ref() {
120                "Tf" => {
121                    let current_font = operation
122                        .operands
123                        .first()
124                        .ok_or_else(|| Error::Syntax("missing font operand".to_string()))?
125                        .as_name()?;
126                    current_encoding = encodings.get(current_font).map(std::string::String::as_str);
127                }
128                "Tj" => {
129                    for bytes in operation.operands.iter_mut().flat_map(Object::as_str_mut) {
130                        let decoded_text = Document::decode_text(current_encoding, bytes);
131                        info!("{}", decoded_text);
132                        if decoded_text == text {
133                            let encoded_bytes = Document::encode_text(current_encoding, other_text);
134                            *bytes = encoded_bytes;
135                        }
136                    }
137                }
138                _ => {}
139            }
140        }
141        let modified_content = content.encode()?;
142        self.change_page_content(page_id, modified_content)
143    }

145    pub fn insert_image(
146        &mut self, page_id: ObjectId, img_object: Stream, position: (f32, f32), size: (f32, f32),
147    ) -> Result<()> {

152
153        let mut content = self.get_and_decode_page_content(page_id)?;
154        content.operations.push(Operation::new("q", vec![]));
155        content.operations.push(Operation::new(
156            "cm",
157            vec![
158                size.0.into(),
159                0.into(),
160                0.into(),
161                size.1.into(),
162                position.0.into(),
163                position.1.into(),
164            ],
165        ));
166        content
167            .operations
168            .push(Operation::new("Do", vec![Name(img_name.as_bytes().to_vec())]));
169        content.operations.push(Operation::new("Q", vec![]));
170
171        self.change_page_content(page_id, content.encode()?)
172    }

193/// Decode CrossReferenceStream
194pub fn decode_xref_stream(mut stream: Stream) -> Result<(Xref, Dictionary)> {
195    stream.decompress();
196    let mut dict = stream.dict;
197    let mut reader = Cursor::new(stream.content);
198    let size = dict
199        .get(b"Size")
200        .and_then(Object::as_i64)
201        .map_err(|_| Error::Xref(XrefError::Parse))?;
202    let mut xref = Xref::new(size as u32, XrefType::CrossReferenceStream);
203    {
204        let section_indice = dict
205            .get(b"Index")
206            .and_then(parse_integer_array)
207            .unwrap_or_else(|_| vec![0, size]);
208        let field_widths = dict
209            .get(b"W")
210            .and_then(parse_integer_array)
211            .map_err(|_| Error::Xref(XrefError::Parse))?;
212
213        if field_widths.len() < 3
214            || field_widths[0].is_negative()
215            || field_widths[1].is_negative()
216            || field_widths[2].is_negative()
217        {
218            return Err(Error::Xref(XrefError::Parse));
219        }
220
221        let mut bytes1 = vec![0_u8; field_widths[0] as usize];
222        let mut bytes2 = vec![0_u8; field_widths[1] as usize];
223        let mut bytes3 = vec![0_u8; field_widths[2] as usize];
224
225        for i in 0..section_indice.len() / 2 {
226            let start = section_indice[2 * i];
227            let count = section_indice[2 * i + 1];
228
229            for j in 0..count {
230                let entry_type = if !bytes1.is_empty() {
231                    read_big_endian_integer(&mut reader, bytes1.as_mut_slice())?
232                } else {
233                    1
234                };
235                match entry_type {
236                    0 => {
237                        //free object
238                        read_big_endian_integer(&mut reader, bytes2.as_mut_slice())?;
239                        read_big_endian_integer(&mut reader, bytes3.as_mut_slice())?;
240                    }
241                    1 => {
242                        //normal object
243                        let offset = read_big_endian_integer(&mut reader, bytes2.as_mut_slice())?;
244                        let generation = if !bytes3.is_empty() {
245                            read_big_endian_integer(&mut reader, bytes3.as_mut_slice())?
246                        } else {
247                            0
248                        } as u16;
249                        xref.insert((start + j) as u32, XrefEntry::Normal { offset, generation });
250                    }
251                    2 => {
252                        //compressed object
253                        let container = read_big_endian_integer(&mut reader, bytes2.as_mut_slice())?;
254                        let index = read_big_endian_integer(&mut reader, bytes3.as_mut_slice())? as u16;
255                        xref.insert((start + j) as u32, XrefEntry::Compressed { container, index });
256                    }
257                    _ => {}
258                }
259            }
260        }
261    }
262    dict.remove(b"Length");
263    dict.remove(b"W");
264    dict.remove(b"Index");
265    Ok((xref, dict))
266}

268fn read_big_endian_integer(reader: &mut Cursor<Vec<u8>>, buffer: &mut [u8]) -> Result<u32> {
269    reader.read_exact(buffer)?;
270    let mut value = 0;
271    for &mut byte in buffer {
272        value = (value << 8) + u32::from(byte);
273    }
274    Ok(value)
275}

277fn parse_integer_array(array: &Object) -> Result<Vec<i64>> {
278    let array = array.as_array()?;
279    let mut out = Vec::with_capacity(array.len());
280
281    for n in array {
282        out.push(n.as_i64()?);
283    }
284
285    Ok(out)
286}
