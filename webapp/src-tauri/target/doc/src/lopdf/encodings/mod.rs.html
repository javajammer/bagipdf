6pub fn bytes_to_string(encoding: [Option<u16>; 256], bytes: &[u8]) -> String {
7    let code_points = bytes
8        .iter()
9        .filter_map(|&byte| encoding[byte as usize])
10        .collect::<Vec<u16>>();
11    String::from_utf16_lossy(&code_points)
12}
13
14pub fn string_to_bytes(encoding: [Option<u16>; 256], text: &str) -> Vec<u8> {
15    text.encode_utf16()
16        .filter_map(|ch| encoding.iter().position(|&code| code == Some(ch)))
17        .map(|byte| byte as u8)
18        .collect()
19}
