26impl fmt::Display for DecryptionError {

30            DecryptionError::MissingRevision => write!(f, "missing encryption revision"),
31            DecryptionError::MissingOwnerPassword => write!(f, "missing the owner password (/O)"),
32            DecryptionError::MissingPermissions => write!(f, "missing the permissions field (/P)"),
33            DecryptionError::InvalidKeyLength => write!(f, "unsupported key length"),
34            DecryptionError::InvalidRevision => write!(f, "unsupported revision"),
35            DecryptionError::InvalidType => write!(f, "unexpected type; document does not comply with the spec"),
36            DecryptionError::MissingFileID => write!(f, "missing the file /ID elements"),
37            DecryptionError::NotDecryptable => write!(f, "the object is not capable of being decrypted"),
38            DecryptionError::IncorrectPassword => write!(f, "the supplied password is incorrect"),
39            DecryptionError::UnsupportedEncryption => {
40                write!(f, "the document uses an encryption scheme that is not supported")
41            }
42        }
43    }

46const PAD_BYTES: [u8; 32] = [
47    0x28, 0xBF, 0x4E, 0x5E, 0x4E, 0x75, 0x8A, 0x41, 0x64, 0x00, 0x4E, 0x56, 0xFF, 0xFA, 0x01, 0x08, 0x2E, 0x2E, 0x00,
48    0xB6, 0xD0, 0x68, 0x3E, 0x80, 0x2F, 0x0C, 0xA9, 0xFE, 0x64, 0x53, 0x69, 0x7A,
49];

51const DEFAULT_KEY_LEN: Object = Object::Integer(40);
52const DEFAULT_ALGORITHM: Object = Object::Integer(0);

54/// Generates the encryption key for the document and, if `check_password` is
55///  true, verifies that the key is correct.
56pub fn get_encryption_key<P>(doc: &Document, password: P, check_password: bool) -> Result<Vec<u8>, DecryptionError>
57where
58    P: AsRef<[u8]>,
59{
60    let password = password.as_ref();
61
62    let encryption_dict = doc
63        .get_encrypted()
64        .map_err(|_| DecryptionError::MissingEncryptDictionary)?;
65
66    // Very early versions of PDF assume a key length of 40 bits
67    let key_len = encryption_dict
68        .get(b"Length")
69        .unwrap_or(&DEFAULT_KEY_LEN)
70        .as_i64()
71        .map_err(|_| DecryptionError::InvalidType)? as usize
72        / 8; // Length is in bits, convert to bytes
73
74    // MD5 produces 128bit digests, so key_len must not be greater
75    if key_len > 128 / 8 {
76        return Err(DecryptionError::InvalidKeyLength);
77    }
78
79    // Make sure we support the encryption algorithm
80    let algorithm = encryption_dict
81        .get(b"V")
82        .unwrap_or(&DEFAULT_ALGORITHM)
83        .as_i64()
84        .map_err(|_| DecryptionError::InvalidType)?;
85    // Currently only support V = 1 or 2
86    if !(1..=2).contains(&algorithm) {
87        return Err(DecryptionError::UnsupportedEncryption);
88    }
89
90    // Revision number dictates hashing strategy
91    let revision = encryption_dict
92        .get(b"R")
93        .map_err(|_| DecryptionError::MissingRevision)?
94        .as_i64()
95        .map_err(|_| DecryptionError::InvalidType)?;
96    if !(2..=3).contains(&revision) {
97        return Err(DecryptionError::UnsupportedEncryption);
98    }
99
100    // Algorithm 3.2
101
102    // 3.2.1 Start building up the key, starting with the user password plaintext,
103    //  padding as needed to 32 bytes
104    let mut key = Vec::with_capacity(128);
105    let password_len = std::cmp::min(password.len(), 32);
106    key.extend_from_slice(&password[0..password_len]);
107    key.extend_from_slice(&PAD_BYTES[0..32 - password_len]);
108
109    // 3.2.3 Append hashed owner password
110    let hashed_owner_password = encryption_dict
111        .get(b"O")
112        .map_err(|_| DecryptionError::MissingOwnerPassword)?
113        .as_str()
114        .map_err(|_| DecryptionError::InvalidType)?;
115    key.extend_from_slice(hashed_owner_password);
116
117    // 3.2.4 Append the permissions (4 bytes)
118    let permissions = encryption_dict
119        .get(b"P")
120        // we don't actually care about the permissions but we need the correct
121        //  value to get the correct key
122        .map_err(|_| DecryptionError::MissingPermissions)?
123        .as_i64()
124        .map_err(|_| DecryptionError::InvalidType)? as u32;
125    key.extend_from_slice(&permissions.to_le_bytes());
126
127    // 3.2.5 Append the first element of the file identifier
128    let file_id_0 = doc
129        .trailer
130        .get(b"ID")
131        .map_err(|_| DecryptionError::MissingFileID)?
132        .as_array()
133        .map_err(|_| DecryptionError::InvalidType)?
134        .first()
135        .ok_or(DecryptionError::InvalidType)?
136        .as_str()
137        .map_err(|_| DecryptionError::InvalidType)?;
138    key.extend_from_slice(file_id_0);
139
140    // 3.2.6 Revision >=4
141    if revision >= 4 {
142        key.extend_from_slice(&[0xFF_u8, 0xFF, 0xFF, 0xFF]);
143    }
144
145    // 3.2.7+8
146    // Hash the contents of key and take the first key_len bytes
147    let n_hashes = if revision < 3 { 1 } else { 51 };
148    for _ in 0..n_hashes {
149        let digest = md5::compute(&key);
150        key.truncate(key_len); // only keep the first key_len bytes
151        key.copy_from_slice(&digest[..key_len]);
152    }
153
154    // Check that the password is correct
155    if check_password {
156        let check = compute_user_password(&key, revision, file_id_0);
157        if let Ok(Object::String(expected, _)) = encryption_dict.get(b"U") {
158            // Only first 16 bytes are significant, the rest are arbitrary padding
159            if expected[..16] != check[..16] {
160                return Err(DecryptionError::IncorrectPassword);
161            }
162        }
163    }
164
165    Ok(key)
166}

168fn compute_user_password<K, ID>(key: K, revision: i64, file_id_0: ID) -> Vec<u8>
169where
170    K: AsRef<[u8]>,
171    ID: AsRef<[u8]>,
172{
173    let key = key.as_ref();
174    let encryptor = Rc4::new(key);
175
176    if revision == 2 {
177        // Algorithm 3.4
178        encryptor.decrypt(PAD_BYTES)
179    } else {
180        // Algorithm 3.5
181        // 3.5.2
182        let mut ctx = md5::Context::new();
183        ctx.consume(PAD_BYTES);
184
185        // 3.5.3
186        ctx.consume(file_id_0);
187        let hash = ctx.compute();
188
189        // 3.5.4
190        let mut encrypted_hash = encryptor.encrypt(&hash[..]);
191
192        // 3.5.5
193        let mut temp_key = vec![0; key.len()];
194        for i in 1..=19 {
195            for (in_byte, out_byte) in key.iter().zip(temp_key.iter_mut()) {
196                *out_byte = in_byte ^ (i as u8);
197            }
198            encrypted_hash = Rc4::new(&temp_key).encrypt(encrypted_hash);
199        }
200
201        // 3.5.6
202        encrypted_hash.extend_from_slice(&PAD_BYTES[0..16]);
203
204        encrypted_hash
205    }
206}

208/// Decrypts `obj` and returns the content of the string or stream.
209/// If obj is not an decryptable type, returns the NotDecryptable error.
210pub fn decrypt_object<Key>(key: Key, obj_id: ObjectId, obj: &Object) -> Result<Vec<u8>, DecryptionError>
211where
212    Key: AsRef<[u8]>,
213{
214    let key = key.as_ref();
215    let mut builder = Vec::<u8>::with_capacity(key.len() + 5);
216    builder.extend_from_slice(key.as_ref());
217
218    // Extend the key with the lower 3 bytes of the object number
219    builder.extend_from_slice(&obj_id.0.to_le_bytes()[..3]);
220    // and the lower 2 bytes of the generation number
221    builder.extend_from_slice(&obj_id.1.to_le_bytes()[..2]);
222
223    // Now construct the rc4 key
224    let key_len = std::cmp::min(key.len() + 5, 16);
225    let rc4_key = &md5::compute(builder)[..key_len];
226
227    let encrypted = match obj {
228        Object::String(content, _) => content,
229        Object::Stream(stream) => &stream.content,
230        _ => {
231            return Err(DecryptionError::NotDecryptable);
232        }
233    };
234
235    // Decrypt using the rc4 algorithm
236    Ok(Rc4::new(rc4_key).decrypt(encrypted))
237}

239#[cfg(test)]
240mod tests {
241    use super::*;
242
243    #[test]
244    fn rc4_works() {
245        let cases = [
246            // Key, Plain, Cipher
247            (
248                String::from("Key"),
249                String::from("Plaintext"),
250                String::from("BBF316E8D940AF0AD3"),
251            ),
252            (String::from("Wiki"), String::from("pedia"), String::from("1021BF0420")),
253        ];
254
255        for (key, plain, cipher) in cases {
256            // Reencode cipher from a hex string to a Vec<u8>
257            let cipher = cipher.as_bytes();
258            let mut cipher_bytes = Vec::with_capacity(cipher.len() / 2);
259            for hex_pair in cipher.chunks_exact(2) {
260                cipher_bytes.push(u8::from_str_radix(std::str::from_utf8(hex_pair).unwrap(), 16).unwrap());
261            }
262
263            let decryptor = Rc4::new(key);
264            let decrypted = decryptor.decrypt(&cipher_bytes);
265            assert_eq!(plain.as_bytes(), &decrypted[..]);
266        }
267    }
