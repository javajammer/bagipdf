13pub fn form(boundingbox: Vec<f32>, matrix: Vec<f32>, content: Vec<u8>) -> Stream {
14    let mut dict = Dictionary::new();
15    dict.set("Type", Object::Name(b"XObject".to_vec()));
16    dict.set("Subtype", Object::Name(b"Form".to_vec()));
17    dict.set(
18        "BBox",
19        Object::Array(boundingbox.into_iter().map(Object::Real).collect()),
20    );
21    dict.set("Matrix", Object::Array(matrix.into_iter().map(Object::Real).collect()));
22    let mut xobject = Stream::new(dict, content);
23    // Ignore any compression error.
24    let _ = xobject.compress();
25    xobject
26}

29pub fn image<P: AsRef<Path>>(path: P) -> Result<Stream> {
30    use std::fs::File;
31    use std::io::prelude::*;
32
33    let mut file = File::open(&path)?;
34    let mut buffer = Vec::new();
35    file.read_to_end(&mut buffer)?;
36
37    image_from(buffer)
38}

40#[cfg(feature = "embed_image")]
41pub fn image_from(buffer: Vec<u8>) -> Result<Stream> {
42    let img = image::load_from_memory(buffer.as_ref())?;
43
44    let (width, height) = img.dimensions();
45
46    // It looks like Adobe Illustrator uses a predictor offset of 2 bytes rather than 1 byte as
47    // the PNG specification suggests. This seems to come from the fact that the PNG specification
48    // doesn't allow 4-bit color images (only 8-bit and 16-bit color). With 1-bit, 2-bit and 4-bit
49    // mono images there isn't the same problem because there's only one component.
50    let bits = img.color().bits_per_pixel() / 3;
51
52    let color_space = match img.color() {
53        ColorType::L8 => b"DeviceGray".to_vec(),
54        ColorType::La8 => b"DeviceGray".to_vec(),
55        ColorType::Rgb8 => b"DeviceRGB".to_vec(),
56        ColorType::Rgb16 => b"DeviceRGB".to_vec(),
57        ColorType::La16 => b"DeviceN".to_vec(),
58        ColorType::Rgba8 => b"DeviceN".to_vec(),
59        ColorType::Rgba16 => b"DeviceN".to_vec(),
60        _ => b"Indexed".to_vec(),
61    };
62
63    let mut dict = Dictionary::new();
64    dict.set("Type", Object::Name(b"XObject".to_vec()));
65    dict.set("Subtype", Object::Name(b"Image".to_vec()));
66    dict.set("Width", width);
67    dict.set("Height", height);
68    dict.set("ColorSpace", Object::Name(color_space));
69    dict.set("BitsPerComponent", bits);
70
71    let is_jpeg = match image::guess_format(buffer.as_ref()) {
72        Ok(format) => match format {
73            ImageFormat::Jpeg => true,
74            _ => false,
75        },
76        Err(_) => false,
77    };
78
79    if is_jpeg {
80        dict.set("Filter", Object::Name(b"DCTDecode".to_vec()));
81        Ok(Stream::new(dict, buffer))
82    } else {
83        let mut img_object = Stream::new(dict, img.into_bytes());
84        // Ignore any compression error.
85        let _ = img_object.compress();
86        Ok(img_object)
87    }
88}
